let lastSend = 0; let smooth = 0; function initControls() { // 1. 鍵盤控制 (PC 備用) window.addEventListener('keydown', e => { if (e.key === 'ArrowLeft') moveLane(-1); if (e.key === 'ArrowRight') moveLane(1); }); // 2. 觸控換道 (手機直接玩模式) window.addEventListener('touchstart', e => { if (!gameStarted || settings.ctrl !== 'touch') return; if (e.touches[0].clientX < window.innerWidth / 2) moveLane(-1); else moveLane(1); }); // 3. 體感偵測 (包含「手機直接玩」與「遠端發送數據」) window.addEventListener("deviceorientation", e => { // A. 如果是「本機玩-手機體感」模式，直接在手機上換道 if (gameStarted && settings.ctrl === 'tilt') { handleTilt(e.gamma); } // B. 如果是連線模式 (MQTT)，則負責發送數據到電腦 if (!client) return; const now = Date.now(); if (now - lastSend < 50) return; // 限制頻率 lastSend = now; smooth = smooth * 0.8 + e.gamma * 0.2; client.publish(topic, JSON.stringify({ type: "tilt", value: smooth }), { qos: 0 }); }); }
